cmake_minimum_required(VERSION 3.16)
project(private_traffic_trends_alerts C)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
add_subdirectory(../jiajun_backend ${CMAKE_CURRENT_BINARY_DIR}/jiajun_backend)
add_subdirectory(../yzy_measurement ${CMAKE_CURRENT_BINARY_DIR}/yzy_measurement)
add_library(pt_trend_alert src/trend_alert.c)
target_include_directories(pt_trend_alert PUBLIC include)
target_link_libraries(pt_trend_alert PUBLIC pt_location_store)
file(MAKE_DIRECTORY ${CMAKE_CURRENT_BINARY_DIR}/web)
configure_file(web/dashboard.html ${CMAKE_CURRENT_BINARY_DIR}/web/dashboard.html COPYONLY)
configure_file(web/dashboard.css ${CMAKE_CURRENT_BINARY_DIR}/web/dashboard.css COPYONLY)
configure_file(web/dashboard.js ${CMAKE_CURRENT_BINARY_DIR}/web/dashboard.js COPYONLY)
add_executable(pt_zian src/main.c)
target_link_libraries(pt_zian PRIVATE pt_trend_alert)
target_compile_definitions(pt_zian PRIVATE PT_DATA_PATH="${CMAKE_CURRENT_BINARY_DIR}/web/data.js")
add_executable(pt_pipeline src/pipeline.c)
target_link_libraries(pt_pipeline PRIVATE pt_trend_alert pt_measurement)
if(MSVC)
  target_compile_options(pt_trend_alert PRIVATE /W4)
else()
  target_compile_options(pt_trend_alert PRIVATE -Wall -Wextra -Wpedantic)
endif()
enable_testing()
add_executable(pt_trend_alert_tests tests/test_trend_alert.c)
target_link_libraries(pt_trend_alert_tests PRIVATE pt_trend_alert)
add_test(NAME pt_trend_alert_tests COMMAND pt_trend_alert_tests)
