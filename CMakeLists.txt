cmake_minimum_required(VERSION 3.16)
project(private_traffic_location_store C)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
add_library(pt_location_store src/location_store.c)
target_include_directories(pt_location_store PUBLIC include ../yzy_measurement/include)
add_executable(pt_locations src/main.c)
target_link_libraries(pt_locations PRIVATE pt_location_store)
if(MSVC)
  target_compile_options(pt_location_store PRIVATE /W4)
else()
  target_compile_options(pt_location_store PRIVATE -Wall -Wextra -Wpedantic)
endif()
enable_testing()
add_executable(pt_location_tests tests/test_location_store.c)
target_link_libraries(pt_location_tests PRIVATE pt_location_store)
add_test(NAME pt_location_tests COMMAND pt_location_tests)
