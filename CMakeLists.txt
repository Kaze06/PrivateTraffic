cmake_minimum_required(VERSION 3.16)
project(private_traffic_measurement C)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
add_library(pt_measurement src/measurement.c src/y4m.c)
target_include_directories(pt_measurement PUBLIC include)
if(MSVC)
  target_compile_options(pt_measurement PRIVATE /W4)
else()
  target_compile_options(pt_measurement PRIVATE -Wall -Wextra -Wpedantic)
endif()
add_executable(pt_demo src/main.c)
target_link_libraries(pt_demo PRIVATE pt_measurement)
enable_testing()
add_executable(pt_tests tests/test_measurement.c)
target_link_libraries(pt_tests PRIVATE pt_measurement)
add_test(NAME pt_tests COMMAND pt_tests)
add_executable(pt_make_sample tests/make_sample.c)
